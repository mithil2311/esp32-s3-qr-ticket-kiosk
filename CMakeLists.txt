cmake_minimum_required(VERSION 3.16)
project(qr_ticket_kiosk_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(KIOSK_SANITIZE "Enable address/undefined sanitizers" OFF)
add_library(cjson vendor/cJSON.c)
target_include_directories(cjson PUBLIC vendor)
add_library(kiosk_core firmware/components/ticket_controller/controller.cpp)
foreach(component platform_hal qr_scanner ticket_controller backend_client printer transaction_journal indicators)
 target_include_directories(kiosk_core PUBLIC firmware/components/${component}/include)
endforeach()
target_link_libraries(kiosk_core PUBLIC cjson)
target_compile_options(kiosk_core PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
add_executable(kiosk_host host_tests/host_main.cpp)
target_link_libraries(kiosk_host PRIVATE kiosk_core)
target_compile_options(kiosk_host PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
if(KIOSK_SANITIZE)
 foreach(t cjson kiosk_core kiosk_host)
  target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${t} PUBLIC -fsanitize=address,undefined)
 endforeach()
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} -m pytest -q ${CMAKE_SOURCE_DIR}/host_tests)
set_tests_properties(integration PROPERTIES ENVIRONMENT "KIOSK_HOST=$<TARGET_FILE:kiosk_host>")
